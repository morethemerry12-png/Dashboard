// REMINDERS ENGINE
// ═══════════════════════════════════════════════════════
const ReminderEngine = {
  isDue(r) {
    const d = new Date(r.nextDate);
    const diffDays = (d - NOW) / 86400000;
    return diffDays <= 3; // due within 3 days or overdue
  },
  isOverdue(r) {
    return new Date(r.nextDate) < NOW;
  },
  advance(id) {
    const r = State.reminders.find(x=>x.id===id);
    if(!r) return;
    const d = new Date(r.nextDate);
    const freq = r.frequency||'monthly';
    if(freq==='weekly') d.setDate(d.getDate()+7);
    else if(freq==='monthly') d.setMonth(d.getMonth()+1);
    else if(freq==='quarterly') d.setMonth(d.getMonth()+3);
    else if(freq==='yearly') d.setFullYear(d.getFullYear()+1);
    r.nextDate = d.toISOString().slice(0,10);
    r.lastPaid = today();
    r.paidHistory = [...(r.paidHistory||[]), today()];
    State.save('reminders');
    App.render();
    App.updateBadges();
    toast('✅ Payment marked — next reminder set!');
  },
  getDue() { return State.reminders.filter(r=>ReminderEngine.isDue(r)); },
  requestPermission() {
    if('Notification' in window && Notification.permission==='default') {
      Notification.requestPermission();
    }
  }
};

    const done = State.projects.filter(p=>p.status==='done');
    if(mode==='active') cnt.innerHTML = Pages._projActive(active);
    else if(mode==='done') cnt.innerHTML = Pages._projDone(done);
    else if(mode==='tasks') cnt.innerHTML = Pages._tasks();
  },

  _projActive(projects) {
    if(projects.length===0) return `<div class="empty"><div class="empty-emoji">🚀</div><div class="empty-title">No active projects</div><div class="empty-sub">Start tracking your vibe coding work.</div><button class="btn-full btn-dark" onclick="Forms.project()">+ New Project</button></div>`;
    return projects.map(p => Pages._projCard(p)).join('');
  },

  _projDone(projects) {
    if(projects.length===0) return `<div class="empty"><div class="empty-emoji">🎉</div><div class="empty-title">No completed projects</div><div class="empty-sub">Mark projects as done when you finish.</div></div>`;
    return projects.map(p => Pages._projCard(p, true)).join('');
  },

  _projCard(p, muted=false) {
    const tasks = State.tasks.filter(t=>t.projectId===p.id);
    const doneT = tasks.filter(t=>t.done).length;
    const pct = tasks.length ? Math.round(doneT/tasks.length*100) : null;
    const client = State.clients.find(c=>c.id===p.clientId);
    return `<div class="card card-body" style="margin-bottom:10px;${muted?'opacity:0.7':''}">
      <div style="display:flex;align-items:flex-start;justify-content:space-between;gap:8px;margin-bottom:10px">
        <div style="flex:1">
          <div style="font-size:16px;font-weight:800;color:var(--text);letter-spacing:-0.2px">${esc(p.name)}</div>
          <div style="font-size:13px;color:var(--text2);margin-top:2px">${client?esc(client.name):esc(p.clientName||'')}${p.revenue>0?' · '+fmt(p.revenue):''}${p.monthlyFee>0?' · '+fmt(p.monthlyFee)+'/mo':''}</div>
        </div>
        <div style="display:flex;gap:6px;align-items:center">
          <span class="badge badge-${p.status==='done'?'green':p.status==='paused'?'orange':'blue'}">${p.status||'active'}</span>
          <button class="icon-btn icon-btn-ghost" onclick="Forms.project('${p.id}')">✏️</button>
        </div>
      </div>
      ${pct!==null?`<div style="margin-bottom:10px">
        <div style="display:flex;justify-content:space-between;font-size:12px;color:var(--text3);margin-bottom:5px"><span>${doneT}/${tasks.length} tasks</span><span>${pct}%</span></div>
        <div class="prog-bar"><div class="prog-fill" style="width:${pct}%"></div></div>
      </div>`:''}
      ${p.notes?`<div style="font-size:13px;color:var(--text2);background:var(--bg);padding:10px 12px;border-radius:10px;line-height:1.5;margin-bottom:10px">${esc(p.notes)}</div>`:''}
      <div style="display:flex;gap:8px;flex-wrap:wrap">
        ${p.status!=='done'?`<button class="ibtn ibtn-green" onclick="Actions.completeProject('${p.id}')">✓ Mark Done</button>`:''}
        ${p.status!=='done'?`<button class="ibtn ibtn-ghost" onclick="Pages.addTask('${p.id}','${esc(p.name)}')">+ Task</button>`:''}
        <button class="ibtn ibtn-ghost" onclick="Actions.deleteProject('${p.id}')">🗑</button>
      </div>
    </div>`;
  },

  _tasks() {
    const openTasks = State.tasks.filter(t=>!t.done);
    const doneTasks = State.tasks.filter(t=>t.done);
    let html = `<div style="display:flex;justify-content:space-between;align-items:center;margin-bottom:10px">
      <div style="font-size:14px;color:var(--text2);font-weight:600">${openTasks.length} open · ${doneTasks.length} done</div>
      <button class="ibtn ibtn-dark" onclick="Forms.task()">+ Task</button>
    </div>`;
    if(State.tasks.length===0) return html+`<div class="empty"><div class="empty-emoji">✅</div><div class="empty-title">No tasks yet</div><div class="empty-sub">Add tasks to track your work.</div></div>`;
    html += `<div class="card card-body">`;
    State.tasks.forEach(t => {
      html += `<div class="check-row">
        <div class="checkmark ${t.done?'checked':''}" onclick="Actions.toggleTask('${t.id}')"></div>
        <div class="check-body" style="flex:1">
          <div class="check-title ${t.done?'done':''}">${esc(t.title)}</div>
          <div class="check-sub">${esc(t.project||'General')}${t.due?' · Due '+t.due:''}</div>
        </div>
        <button class="icon-btn icon-btn-ghost" onclick="Actions.deleteTask('${t.id}')" style="margin-left:8px">×</button>
      </div>`;
    });
    html += `</div>`;
    return html;
  },

  addTask(projectId, projectName) {
    Forms.task(projectId, projectName);
  },

  // ── REMINDERS ─────────────────────────────────────────
  reminders() {
    const due = ReminderEngine.getDue();
    const upcoming = State.reminders.filter(r=>!ReminderEngine.isDue(r));

    let html = '';

    if(State.reminders.length===0) return `<div class="empty">
      <div class="empty-emoji">🔔</div>
      <div class="empty-title">No reminders set</div>
      <div class="empty-sub">Set up recurring payment reminders for your clients and never miss a collection.</div>
      <button class="btn-full btn-dark" onclick="Forms.reminder()">+ Set First Reminder</button>
    </div>`;

    if(due.length>0) {
      html += `<div class="section-head"><div class="section-title">Action Required</div></div>`;
      due.forEach(r => {
        html += `<div class="reminder-card ${ReminderEngine.isOverdue(r)?'overdue':'due'}">
          <div style="display:flex;justify-content:space-between;align-items:flex-start;margin-bottom:10px">
            <div>
              <div style="font-size:16px;font-weight:800;color:var(--text)">${esc(r.clientName)}</div>
              <div style="font-size:13px;color:var(--text2);margin-top:2px">${fmt(r.amount)} · ${r.frequency}</div>
            </div>
            <span class="badge badge-${ReminderEngine.isOverdue(r)?'red':'orange'}">${ReminderEngine.isOverdue(r)?'Overdue':'Due now'}</span>
          </div>
          ${r.notes?`<div style="font-size:13px;color:var(--text2);margin-bottom:10px">${esc(r.notes)}</div>`:''}
          <div style="display:flex;gap:8px">
            <button class="ibtn ibtn-dark" style="flex:1;justify-content:center" onclick="ReminderEngine.advance('${r.id}')">✓ Mark as Paid</button>
            <button class="ibtn ibtn-ghost" onclick="Forms.reminder('${r.id}')">✏️</button>
            <button class="ibtn ibtn-ghost" onclick="Actions.deleteReminder('${r.id}')">🗑</button>
          </div>
        </div>`;
      });
      if(upcoming.length>0) html += `<div class="divider"></div>`;
    }

    if(upcoming.length>0) {
      html += `<div class="section-head"><div class="section-title">Upcoming</div></div>`;
      upcoming.forEach(r => {
        const nextDate = new Date(r.nextDate);
        const daysUntil = Math.ceil((nextDate-NOW)/86400000);
        html += `<div class="reminder-card">
          <div style="display:flex;justify-content:space-between;align-items:flex-start;margin-bottom:8px">
            <div>
              <div style="font-size:15px;font-weight:700;color:var(--text)">${esc(r.clientName)}</div>
              <div style="font-size:13px;color:var(--text2);margin-top:1px">${fmt(r.amount)} · ${r.frequency}</div>
            </div>
            <div style="text-align:right">
              <div style="font-size:13px;font-weight:700;color:var(--text2)">in ${daysUntil}d</div>
              <div style="font-size:11px;color:var(--text3)">${nextDate.toLocaleDateString('en-GB',{day:'numeric',month:'short'})}</div>
            </div>
          </div>
          ${r.lastPaid?`<div style="font-size:12px;color:var(--text3);margin-bottom:8px">Last paid: ${r.lastPaid}</div>`:''}
          <div style="display:flex;gap:8px">
            <button class="ibtn ibtn-ghost" onclick="Forms.reminder('${r.id}')">✏️ Edit</button>
            <button class="ibtn ibtn-green" onclick="ReminderEngine.advance('${r.id}')">Paid early ✓</button>
            <button class="ibtn ibtn-ghost" onclick="Actions.deleteReminder('${r.id}')">🗑</button>
          </div>
        </div>`;
      });
    }

    return html;
  },

  // ── REVENUE ───────────────────────────────────────────
  revenue() {
    const total = State.revenue.reduce((s,r)=>s+(parseFloat(r.amount)||0),0);
    const thisMonth = State.revenue.filter(r=>{
      const d=new Date(r._ts||0); return d.getMonth()===NOW.getMonth()&&d.getFullYear()===NOW.getFullYear();
    }).reduce((s,r)=>s+(parseFloat(r.amount)||0),0);
    const mrr = State.clients.reduce((s,c)=>s+(parseFloat(c.monthlyFee)||0),0);

    let html = `<div class="stats-row" style="grid-template-columns:1fr 1fr 1fr;margin-bottom:16px">
      <div class="stat-card"><div class="stat-label">All Time</div><div class="stat-value" style="font-size:20px">${fmt(total)}</div></div>
      <div class="stat-card"><div class="stat-label">This Month</div><div class="stat-value" style="font-size:20px">${fmt(thisMonth)}</div></div>
      <div class="stat-card"><div class="stat-label">MRR</div><div class="stat-value" style="font-size:20px">${fmt(mrr)}</div></div>
    </div>`;

    if(State.revenue.length===0) return html+`<div class="empty"><div class="empty-emoji">💰</div><div class="empty-title">No revenue logged</div><div class="empty-sub">Start logging payments to track your earnings.</div><button class="btn-full btn-green" onclick="Forms.revenue()">+ Log First Payment</button></div>`;

    html += `<div class="section-head"><div class="section-title">Payment History</div></div>`;
    html += `<div class="card">`;
    State.revenue.forEach(r => {
      html += `<div class="list-item">
        <div class="list-item-icon" style="background:var(--green-light);font-size:18px">💵</div>
        <div class="list-item-body">
          <div class="list-item-title">${esc(r.label||'Payment')}</div>
          <div class="list-item-sub">${esc(r.clientName||'')}${r.date?' · '+r.date:''}</div>
        </div>
        <div class="list-item-right">
          <span style="font-size:16px;font-weight:800;color:var(--green)">+${fmt(r.amount)}</span>
        </div>
        <button class="icon-btn icon-btn-ghost" onclick="Actions.deleteRevenue('${r.id}')" style="margin-left:4px">×</button>
      </div>`;
    });
    html += `</div>`;
    return html;
  }
};

const Actions = {
  saveClient(editId) {
    const name = ($('#sheet-body #f-name')||{}).value?.trim();
    if(!name) { toast('⚠️ Client name is required'); return; }
    const data = {
      name,
      phone: ($('#sheet-body #f-phone')||{}).value?.trim()||'',
      email: ($('#sheet-body #f-email')||{}).value?.trim()||'',
      website: ($('#sheet-body #f-website')||{}).value?.trim()||'',
      payment: parseFloat(($('#sheet-body #f-payment')||{}).value)||0,
      monthlyFee: parseFloat(($('#sheet-body #f-monthly')||{}).value)||0,
      status: ($('#sheet-body #f-status')||{}).value||'active',
      notes: ($('#sheet-body #f-notes')||{}).value?.trim()||'',
    };
    if(editId) {
      const idx = State.clients.findIndex(x=>x.id===editId);
      if(idx>-1) State.clients[idx] = {...State.clients[idx],...data};
    } else {
      const newClient = {id:uid(),...data};
      State.clients.unshift(newClient);
      // Auto-create reminder if monthly fee
      if(data.monthlyFee>0) {
        const nextDate = new Date(); nextDate.setMonth(nextDate.getMonth()+1);
        State.reminders.unshift({id:uid(),clientName:name,amount:data.monthlyFee,frequency:'monthly',nextDate:nextDate.toISOString().slice(0,10),notes:'Monthly fee',paidHistory:[]});
        State.save('reminders');
        toast('🔔 Reminder auto-created for monthly fee!', 3000);
      }
    }
    State.save('clients');
    Sheet.forceClose();
    App.render(); App.updateBadges();
    if(!editId) toast('✅ Client added!');
    else toast('✅ Client updated!');
  },

  saveProject(editId) {
    const name = ($('#sheet-body #f-name')||{}).value?.trim();
    if(!name) { toast('⚠️ Project name is required'); return; }
    const clientSel = ($('#sheet-body #f-client')||{}).value||'';
    const client = State.clients.find(c=>c.id===clientSel);
    const revenue = parseFloat(($('#sheet-body #f-revenue')||{}).value)||0;
    const data = {
      name,
      clientId: clientSel,
      clientName: client?client.name:'',
      revenue,
      monthlyFee: parseFloat(($('#sheet-body #f-monthly')||{}).value)||0,
      status: ($('#sheet-body #f-status')||{}).value||'active',
      notes: ($('#sheet-body #f-notes')||{}).value?.trim()||'',
    };
    if(editId) {
      const idx = State.projects.findIndex(x=>x.id===editId);
      if(idx>-1) State.projects[idx] = {...State.projects[idx],...data};
    } else {
      State.projects.unshift({id:uid(),date:today(),...data});
      if(revenue>0) {
        State.revenue.unshift({id:uid(),amount:revenue,label:name,clientId:clientSel,clientName:client?client.name:'',date:today(),_ts:Date.now()});
        State.save('revenue');
      }
    }
    State.save('projects');
    Sheet.forceClose();
    App.render(); App.updateBadges();
    toast(editId?'✅ Project updated!':'🚀 Project added!');
  },

  saveReminder(editId) {
    const sel = ($('#sheet-body #f-client')||{}).value||'';
    const clientName = sel==='__custom' ? (($('#sheet-body #f-custom')||{}).value?.trim()||'') : sel;
    const amount = parseFloat(($('#sheet-body #f-amount')||{}).value)||0;
    if(!clientName) { toast('⚠️ Please select a client'); return; }
    if(!amount) { toast('⚠️ Please enter an amount'); return; }
    const dateVal = ($('#sheet-body #f-date')||{}).value;
    const data = {
      clientName,
      amount,
      frequency: ($('#sheet-body #f-freq')||{}).value||'monthly',
      nextDate: dateVal||todayISO(),
      notes: ($('#sheet-body #f-notes')||{}).value?.trim()||'',
      paidHistory: editId ? (State.reminders.find(x=>x.id===editId)||{}).paidHistory||[] : [],
    };
    if(editId) {
      const idx = State.reminders.findIndex(x=>x.id===editId);
      if(idx>-1) State.reminders[idx] = {...State.reminders[idx],...data};
    } else {
      State.reminders.unshift({id:uid(),...data});
      ReminderEngine.requestPermission();
    }
    State.save('reminders');
    Sheet.forceClose();
    App.render(); App.updateBadges();
    toast('🔔 Reminder saved!');
  },

  saveRevenue() {
    const amount = parseFloat(($('#sheet-body #f-amount')||{}).value)||0;
    if(!amount) { toast('⚠️ Please enter an amount'); return; }
    const clientSel = ($('#sheet-body #f-client')||{}).value||'';
    const [clientId, clientName] = clientSel.includes('|') ? clientSel.split('|') : ['',''];
    State.revenue.unshift({
      id:uid(),
      amount,
      label: ($('#sheet-body #f-label')||{}).value?.trim()||'Payment',
      clientId, clientName,
      date: today(),
      _ts: Date.now(),
    });
    State.save('revenue');
    Sheet.forceClose();
    App.render(); App.updateBadges();
    toast('💰 Revenue logged!');
  },

  saveTask() {
    const title = ($('#sheet-body #f-title')||{}).value?.trim();
    if(!title) { toast('⚠️ Task title is required'); return; }
    const projSel = ($('#sheet-body #f-project')||{}).value||'';
    const [projectId, projectName] = projSel.includes('|') ? projSel.split('|') : ['',''];
    State.tasks.unshift({
      id:uid(), title,
      projectId, project:projectName,
      due: ($('#sheet-body #f-due')||{}).value||'',
      done:false,
    });
    State.save('tasks');
    Sheet.forceClose();
    App.render(); App.updateBadges();
    toast('✅ Task added!');
  },

  toggleTask(id) {
    const t = State.tasks.find(x=>x.id===id);
    if(!t) return;
    t.done = !t.done;
    State.save('tasks');
    App.render(); App.updateBadges();
    if(t.done) toast('✅ Task complete!');
  },

  completeProject(id) {
    const p = State.projects.find(x=>x.id===id);
    if(!p) return;
    p.status = 'done';
    State.save('projects');
    App.render(); App.updateBadges();
    toast('🎉 Project completed!');
  },

  deleteClient(id) {
    if(!confirm('Remove this client?')) return;
    State.clients = State.clients.filter(x=>x.id!==id);
    State.save('clients');
    Sheet.forceClose();
    App.render(); App.updateBadges();
    toast('🗑 Client removed');
  },

  deleteProject(id) {
    if(!confirm('Remove this project?')) return;
    State.projects = State.projects.filter(x=>x.id!==id);
    State.save('projects');
    App.render(); App.updateBadges();
    toast('🗑 Project removed');
  },

  deleteReminder(id) {
    State.reminders = State.reminders.filter(x=>x.id!==id);
    State.save('reminders');
    App.render(); App.updateBadges();
    toast('🗑 Reminder removed');
  },

  deleteRevenue(id) {
    State.revenue = State.revenue.filter(x=>x.id!==id);
    State.save('revenue');
    App.render(); App.updateBadges();
    toast('🗑 Entry removed');
  },

  deleteTask(id) {
    State.tasks = State.tasks.filter(x=>x.id!==id);
    State.save('tasks');
    App.render(); App.updateBadges();
  },
};
